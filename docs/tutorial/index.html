1.2 useState Hook

useState是React中最基础的Hook，用于在函数组件中添加状态。

基础用法 tsx 
import { useState } from 'react'

function Counter() {
  // 声明状态变量 count 和更新函数 setCount
  const [count, setCount] = useState(0)

  return (
    <div>
      <p>当前计数: {count}</p>
      <button onClick={() => setCount(count + 1)}>
        增加
      </button>
    </div>
  )
}
💡 实战应用

在登录页面中，我们使用useState管理loading状态：

src/views/login/Login.tsx - 登录状态管理 tsx 
import { useState } from 'react'
import { Button, Form, Input, message } from 'antd'
import api from '@/api'

export default function LoginFC() {
  // 使用useState管理登录loading状态
  const [loading, setLoading] = useState(false)
  const updateToken = useStore(state => state.updateToken)

  const onFinish = async (values: Login.params) => {
    try {
      setLoading(true)  // 开始加载
      const data = await api.login(values)
      setLoading(false) // 结束加载
      storage.set('token', data)
      updateToken(data)
      message.success('登录成功')
    } catch (error) {
      setLoading(false) // 错误时也要关闭loading
    }
  }

  return (
    <Button type="primary" loading={loading}>
      登录
    </Button>
  )
}

1.3 useEffect Hook

useEffect用于处理副作用，如数据获取、订阅、手动DOM操作等。

useEffect语法详解 tsx 
import { useEffect } from 'react'

function UserProfile({ userId }) {
  const [user, setUser] = useState(null)

  // 依赖数组为空：只在组件挂载时执行一次
  useEffect(() => {
    fetchUser(userId).then(setUser)
  }, [])

  // 依赖userId：userId变化时重新执行
  useEffect(() => {
    if (userId) {
      fetchUser(userId).then(setUser)
    }
  }, [userId])

  // 无依赖数组：每次渲染都执行（慎用）
  useEffect(() => {
    console.log('每次渲染都会执行')
  })

  return <div>{user?.name}</div>
}

1.4 useContext 和 useReducer

这两个Hook通常组合使用，实现全局状态管理（类似Vuex）。

Context + Reducer模式 tsx 
import { createContext, useContext, useReducer } from 'react'

// 创建Context
const StoreContext = createContext()

// Reducer函数
function reducer(state, action) {
  switch (action.type) {
    case 'INCREMENT':
      return { ...state, count: state.count + 1 }
    case 'DECREMENT':
      return { ...state, count: state.count - 1 }
    default:
      return state
  }
}

// Provider组件
export function StoreProvider({ children }) {
  const [state, dispatch] = useReducer(reducer, { count: 0 })
  
  return (
    <StoreContext.Provider value={{ state, dispatch }}>
      {children}
    </StoreContext.Provider>
  )
}

// 自定义Hook
export function useStore() {
  return useContext(StoreContext)
}

1.5 useMemo 和 useCallback

用于性能优化，避免不必要的重复计算和渲染。

性能优化Hooks tsx 
import { useMemo, useCallback } from 'react'

function ExpensiveComponent({ list, onItemClick }) {
  // useMemo: 缓存计算结果
  const sortedList = useMemo(() => {
    console.log('执行排序计算')
    return [...list].sort((a, b) => a.price - b.price)
  }, [list]) // 仅当list变化时重新计算

  // useCallback: 缓存函数引用
  const handleClick = useCallback((id) => {
    console.log('点击项目:', id)
    onItemClick(id)
  }, [onItemClick]) // 仅当onItemClick变化时创建新函数

  return (
    <ul>
      {sortedList.map(item => (
        <li key={item.id} onClick={() => handleClick(item.id)}>
          {item.name}
        </li>
      ))}
    </ul>
  )
}
⚠️ 注意事项

不要过度优化！只有在以下情况才使用memo/useMemo/useCallback：

  • 组件渲染开销大
  • 状态频繁更新导致子组件重复渲染
  • 函数作为props传递给React.memo包裹的组件

1.6 useRef

useRef用于访问DOM元素或保存可变值而不触发重新渲染。

useRef两种用途 tsx 
import { useRef } from 'react'

function MyComponent() {
  // 1. 访问DOM元素
  const inputRef = useRef(null)
  
  // 2. 保存可变值（不触发重渲染）
  const timerRef = useRef(null)

  const focusInput = () => {
    inputRef.current?.focus()
  }

  const startTimer = () => {
    timerRef.current = setInterval(() => {
      console.log('定时器运行中...')
    }, 1000)
  }

  return (
    <div>
      <input ref={inputRef} placeholder="输入框" />
      <button onClick={focusInput}>聚焦</button>
    </div>
  )
}

2.1 基础类型

TypeScript在JavaScript基础上添加了静态类型系统，让我们在编写代码时就能发现潜在错误。

常用基础类型 typescript 
// 布尔值
let isDone: boolean = false

// 数字
let age: number = 25

// 字符串
let name: string = '张三'

// 数组
let list: number[] = [1, 2, 3]
let names: Array<string> = ['Alice', 'Bob']

// 元组 - 固定长度和类型的数组
let tuple: [string, number] = ['age', 25]

// 枚举
enum Color {
  Red = 'RED',
  Green = 'GREEN',
  Blue = 'BLUE'
}

// Any - 任意类型（尽量少用）
let notSure: any = 4
notSure = 'maybe a string'

// Void - 无返回值
function warnUser(): void {
  console.log('This is a warning')
}

// Null 和 Undefined
let u: undefined = undefined
let n: null = null

// Never - 永不存在的值的类型
function error(message: string): never {
  throw new Error(message)
}

2.2 接口 (Interface)

接口是TypeScript中定义对象形状的核心工具，在项目中使用频率极高。

接口定义与使用 typescript

interface User {

  name: string

  createdAt: new Date()

// 函数类型接口
interface SearchFunc {

// 接口继承

}

2.3 泛型 (Generics)

泛型允许我们创建可重用的、适用于多种类型的组件和函数。

泛型基础与进阶 typescript 
// 基础泛型函数
function identity<T>(arg: T): T {
  return arg
}

// 使用泛型
let output = identity<string>('Hello')
let output2 = identity<number>(123)

// 泛型接口
interface ApiResponse<T> {
  code: number
  data: T
  message: string
}

// 泛型约束
interface Lengthwise {
  length: number
}

function loggingIdentity<T extends Lengthwise>(arg: T): T {
  console.log(arg.length)
  return arg
}

// 多个泛型参数
function pair<T, U>(first: T, second: U): [T, U] {
  return [first, second]
}

const myPair = pair<string, number>('age', 25)

3.1 Vite构建工具

Vite是下一代前端构建工具，相比Webpack具有更快的启动速度和热更新速度。

💡 Vite优势
  • ⚡️ 极速的服务启动 - 基于浏览器原生ES modules
  • 🔥 轻量快速的热重载 - 无论应用大小，HMR速度始终如一
  • 🛠️ 丰富的功能 - 内置TypeScript、JSX、CSS等支持
  • 📦 优化的构建 - 基于Rollup，输出高效的生产代码
vite.config.ts - Vite配置文件详解 typescript 
import { defineConfig } from 'vite'
import react from '@vitejs/plugin-react'
import path from 'path'

export default defineConfig({
  // 插件配置
  plugins: [react()],
  
  // 路径别名配置
  resolve: {
    alias: {
      '@': path.resolve(__dirname, './src')
    }
  },
  
  // 开发服务器配置
  server: {
    port: 3000,  // 端口号
    open: true,   // 自动打开浏览器
    proxy: {
      // API代理配置 - 解决跨域问题
      '/api': {
        target: 'http://localhost:8080',
        changeOrigin: true,
        rewrite: (path) => path.replace(/^\/api/, '')
      }
    }
  },
  
  // 构建配置
  build: {
    outDir: 'dist',
    sourcemap: false,
    rollupOptions: {
      output: {
        manualChunks: {
          // 代码分割 - vendor包
          vendor: ['react', 'react-dom'],
          // antd单独打包
          antd: ['antd']
        }
      }
    }
  }
})

5.1 Axios实例创建

src/utils/request.ts - Axios封装 typescript 
import axios, { AxiosError } from 'axios'
import { showLoading, hideLoading } from './loading'
import storage from './storage'
import env from '@/config'

// 创建实例
const instance = axios.create({
  timeout: 8000,
  timeoutErrorMessage: '请求超时，请稍后再试',
  withCredentials: true,
  headers: {
    icode: ''
  }
})

// 请求拦截器
instance.interceptors.request.use(
  config => {
    if (config.showLoading) showLoading()
    
    // 添加Token
    const token = storage.get('token')
    if (token) {
      config.headers.Authorization = 'Bearer ' + token
    }
    
    // 环境配置
    if (env.mock) {
      config.baseURL = env.mockApi
    } else {
      config.baseURL = env.baseApi
    }
    
    return config
  },
  (error: AxiosError) => {
    return Promise.reject(error)
  }
)

// 响应拦截器
instance.interceptors.response.use(
  response => {
    const data: Result = response.data
    hideLoading()
    
    // Token过期处理
    if (data.code === 500001) {
      message.error(data.msg)
      storage.remove('token')
      location.href = '/login?callback=' + encodeURIComponent(location.href)
    } else if (data.code != 0) {
      message.error(data.msg)
      return Promise.reject(data)
    }
    
    return data.data as never
  },
  error => {
    hideLoading()
    message.error(error.message)
    return Promise.reject(error.message)
  }
)
